### 2.1 include 规则

```
APP   → #include "api_*.h", "msg_scheduler.h", "msg_core.h"
API   → #include "proto_*.h", "drv_*.h", "msg_scheduler.h"
PROTO → #include "drv_*.h", "hal_*.h"
DRV   → #include "hal_*.h"
HAL   → #include <MCU寄存器头文件>
CORE  → 零外部依赖
```

上层可引用下层，同层不互引，所有业务通信走消息调度器。

## 四、消息体系

### 4.1 核心原则

**msg_def.h 不定义任何业务结构体。** 调度器只传递指针，不关心内容。

```
msg_def.h        → MsgId_t + Msg_t(id + param + void*data_ptr)
msg_scheduler    → 只操作 Msg_t，永不 dereference data_ptr
各模块自己的 .h  → 定义该模块进出消息的结构体（归模块所有）
interface_registry.h → 不被编译，纯文档，列出全系统消息契约
```

### 4.2 Msg_t（通用消息结构）

```c
// msg_def.h —— 只有这3个定义
typedef enum {
    MSG_KEY_EVENT,
    MSG_COOKING_CTRL,
    MSG_POWER_CTRL,
    MSG_FAN_CTRL,
    MSG_DISPLAY_REFRESH,
    MSG_TIMER_100MS,
    MSG_TIMER_1S,
    MSG_COMM_TX_DONE,
    MSG_COMM_DATA_UPDATE,
    MSG_SYSTEM_ERROR,
    MSG_COUNT
} MsgId_t;

typedef struct {
    MsgId_t id;
    uint16_t param;       // 轻量数值（键码、档位…），复杂数据走data_ptr
    void   *data_ptr;     // 指向模块自定义结构体，调度器不解包
} Msg_t;

// 调度器回调签名
typedef void (*MsgHandler_t)(MsgId_t id, uint16_t param, void *data_ptr);
```

### 4.3 各模块自行定义的结构体（归模块 .h 所有）

```c
// ---- api_comm.h / proto_modbus.h ----
#pragma pack(1)
typedef struct {
    uint8_t  head_id;           // 炉头号 0-3
    uint16_t igbtTemp;          // IGBT温度 (定点 0.01°C)
    uint16_t surfaceTemp;       // 表面温度 (定点 0.01°C)
    uint16_t coilCurrent;       // 线圈电流 (mA)
    uint16_t powerActual;       // 实际功率 (W)
    uint8_t  systemState;       // 状态字
    uint8_t  faultCode;         // 故障码 0=正常
    uint8_t  commStatus;        // 0=正常 1=超时 2=CRC错
} ModbusData_t;
#pragma pack()

// ---- app_cook.h ----
typedef struct {
    uint8_t  head_id;           // 炉头号
    uint8_t  cmd;               // 0=停止 1=开始 2=暂停 3=恢复
} CookCtrl_t;

// ---- app_power_ctrl.h ----
typedef struct {
    uint8_t  head_id;           // 炉头号
    uint16_t power_watt;        // 目标功率 (W)
    uint8_t  ramp;              // 0=直接 1=爬坡
} PowerCtrl_t;

// ---- app_protect.h ----
typedef struct {
    uint8_t  head_id;
    uint8_t  fault_type;        // 0=无 1=IGBT过温 2=线圈过流 3=表面过温
    uint16_t fault_value;       // 故障时数值
} ProtectFault_t;
```

### 4.4 COOK表格结构（时间+功率+温度）

```c
// 菜谱步骤表（Flash常量）
typedef struct {

    uint8_t  fan_speed;         // 当前风机档位
    uint8_t  timer_enabled;     // 定时使能
    uint16_t timer_remain_sec;  // 剩余秒数
    uint8_t  boost_active;      // Boost激活
    uint32_t boost_elapsed_ms;  // Boost已运行ms (最大300000)
    uint8_t  recipe_id;         // 当前菜谱ID
    uint8_t  step_index;        // 当前步骤索引
    uint16_t step_elapsed_sec;  // 当前步骤已过秒数
```

### 4.5 interface_registry.h（不参与编译，纯契约文档）

```c
/*
 * interface_registry.h —— 全系统消息契约注册表
 *
 * ★ 本文件不被任何 .c include ★
 * 用途：
 *   1. 人审查：一眼看清模块间数据流
 *   2. AI 检查：编译后 grep/脚本对照，检查模块是否违反契约
 *   3. Python 测试：自动生成 mock 输入和断言
 *
 * 格式：MSG_ID  SENDER     RECEIVER          data_ptr 指向类型
 */

REGISTRY_TABLE_BEGIN

// 按键事件（无 data_ptr，仅用 param）
MSG( KEY_EVENT,        api_key,       app_ui,         NULL            )
MSG( KEY_EVENT,        api_key,       app_cook,       NULL            )

// 烹饪控制：app_ui → app_cook
MSG( COOKING_CTRL,     app_ui,        app_cook,       CookCtrl_t*     )

// 功率下发链：cook → power_ctrl → api_comm / actuator
MSG( POWER_CTRL,       app_cook,      app_power_ctrl, PowerCtrl_t*    )
MSG( POWER_CTRL,       app_temp_ctrl, app_power_ctrl, PowerCtrl_t*    )
MSG( POWER_CTRL,       app_protect,   app_power_ctrl, PowerCtrl_t*    )
MSG( POWER_CTRL,       app_power_ctrl,api_comm,       PowerCtrl_t*    )
MSG( POWER_CTRL,       app_power_ctrl,app_actuator,   PowerCtrl_t*    )

// 风机控制
MSG( FAN_CTRL,         app_cook,      app_actuator,   NULL            )

// 显示刷新
MSG( DISPLAY_REFRESH,  app_ui,        api_display,    NULL            )
MSG( DISPLAY_REFRESH,  app_cook,      api_display,    NULL            )

// 定时节拍
MSG( TIMER_100MS,      hal_timer,     app_ui,         NULL            )
MSG( TIMER_100MS,      hal_timer,     app_protect,    NULL            )
MSG( TIMER_100MS,      hal_timer,     api_comm,       NULL            )
MSG( TIMER_100MS,      hal_timer,     api_display,    NULL            )
MSG( TIMER_1S,         hal_timer,     app_cook,       NULL            )
MSG( TIMER_1S,         hal_timer,     app_temp_ctrl,  NULL            )

// MODBUS通信
MSG( COMM_TX_DONE,     api_comm,      app_cook,       NULL            )
MSG( COMM_DATA_UPDATE, api_comm,      app_ui,         ModbusData_t*   )
MSG( COMM_DATA_UPDATE, api_comm,      app_cook,       ModbusData_t*   )
MSG( COMM_DATA_UPDATE, api_comm,      app_temp_ctrl,  ModbusData_t*   )
MSG( COMM_DATA_UPDATE, api_comm,      app_protect,    ModbusData_t*   )

// 系统错误
MSG( SYSTEM_ERROR,     app_protect,   app_ui,         ProtectFault_t* )
MSG( SYSTEM_ERROR,     app_cook,      app_ui,         NULL            )

REGISTRY_TABLE_END
```

### 4.7 DisplayCmd_t（api_display.h 内部定义，非消息结构体）

```
高8位=命令类型, 低8位=参数
  0x01xx = 显示档位    (xx=炉头号, param=档位0-9/B/P)
  0x02xx = 显示横杠闪烁 (xx=炉头号)
  0x03xx = 显示定时剩余 (xx=炉头号, param=剩余分钟)
  0x04xx = 显示组合符号 (xx=符号类型: 0=|, 1=-, 2=□, 3=交错)
  0x05xx = 显示PA暂停  (xx=炉头号)
  0x06xx = 显示错误码  (xx=错误码)
  0x0700 = 全显测试
  0x0800 = 版本显示 "V1.0 / P1.0"
  0x0900 = LED控制    (低8位按位: bit0=电源, bit1=童锁, bit2=无区, ...)
```

## 五、数据流

### 5.1 按键→加热 完整链路

```
触摸中断 → drv_touch扫描 → api_key去抖+长按检测
  → Msg_Post(MSG_KEY_EVENT, keyCode)
    →(1ms后) app_ui回调: 匹配当前节点事件 → 执行规则call
      → Msg_Post(MSG_DISPLAY_REFRESH, cmd)
        →(1ms后) api_display: 更新段码缓冲
      → Msg_Post(MSG_COOKING_CTRL, head|cmd)
        →(1ms后) app_cook: 查表→更新HeadState
          → Msg_Post(MSG_POWER_CTRL, head|targetPower)
            →(1ms后) app_power_ctrl: 功率爬坡→下发最终值
              → api_comm: 加入MODBUS发送队列
              → app_actuator: 加热使能
```

### 5.2 MODBUS回读→保护→温控 链路

```
100ms定时 → api_comm轮询调度
  → proto_modbus: 调用MODBUS库发送查询
    → 库回调返回数据帧→解码→填充CommData_t
      → MsgScheduler_UpdateCommData(&commData)
        → app_protect(优先级最高): IGBT/电流/温度超限?
          ├→ 超限: Msg_Post(MSG_SYSTEM_ERROR) → app_ui显示
          │         Msg_Post(MSG_POWER_CTRL, 0) → 强制停机
          └→ 正常: 放行
        → app_temp_ctrl: 实际温度 vs 目标温度
          → 需调整? Msg_Post(MSG_POWER_CTRL, 修正值)
        → app_ui: 更新温度显示
        → app_cook: 更新工作状态
```

### 5.3 1ms中断调度

```
SysTick_Handler:
  BUZZER_Tick1ms()          // 蜂鸣器状态机（独立于消息系统）
  MsgScheduler_Run1ms()     // 取一条消息→匹配回调→执行
  api_display_scan()        // 数码管段扫描刷新（如有需要）
```
