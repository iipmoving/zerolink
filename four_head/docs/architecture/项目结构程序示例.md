#ifndef MSG_CORE_H
#define MSG_CORE_H
#include <stdint.h>
// ========== 消息ID ==========
typedef enum {
    MSG_KEY_EVENT,          // 按键事件
    MSG_COOKING_CTRL,       // 烹饪控制命令
    MSG_POWER_CTRL,         // 功率/加热控制
    MSG_FAN_CTRL,           // 风机控制
    MSG_DISPLAY_REFRESH,    // 显示刷新
    MSG_TIMER_100MS,        // 100ms节拍
    MSG_COMM_DATA_UPDATE,   // 通讯数据更新通知
    MSG_SYSTEM_ERROR,       // 系统错误
} MsgId_t;
// ========== 消息结构 ==========
typedef struct {
    MsgId_t id;
    uint16_t param;
} Msg_t;
// ========== 通讯数据（调度器内部存储） ==========
typedef struct {
    uint16_t igbtTemp;
    uint16_t surfaceTemp;
    uint16_t coilCurrent;
    uint16_t systemState;
    uint8_t  faultCode;
    uint8_t  commStatus;    // 0=正常, 1=超时, 2=校验错
} CommData_t;
// ========== 回调函数类型 ==========
typedef void (*MsgHandler_t)(Msg_t *msg, void *dataPtr);
#endif

### 3. 调度器实现 (`src/msg_scheduler.c`)

c

#include "msg_scheduler.h"
#define MAX_HANDLERS 12
#define MSG_SLOT_DEPTH 4   // 环形队列深度，实际只用到1个槽
// ========== 消息槽（环形队列） ==========
static Msg_t msgSlot[MSG_SLOT_DEPTH];
static uint8_t slotHead = 0;
static uint8_t slotTail = 0;
static uint8_t slotCount = 0;
// ========== 回调注册表 ==========
typedef struct {
    MsgId_t id;
    MsgHandler_t handler;
} HandlerEntry_t;
static HandlerEntry_t handlers[MAX_HANDLERS];
static uint8_t handlerCount = 0;
// ========== 内部通讯数据存储 ==========
static CommData_t commData;
// ========== 初始化 ==========
void MsgScheduler_Init(void) {
    slotHead = 0;
    slotTail = 0;
    slotCount = 0;
    handlerCount = 0;
    // 通讯数据初始化为无效值
    commData.igbtTemp = 0xFFFF;
    commData.surfaceTemp = 0xFFFF;
    commData.coilCurrent = 0;
    commData.systemState = 0;
    commData.faultCode = 0;
    commData.commStatus = 0;
}
// ========== 注册 ==========
void MsgScheduler_Register(MsgId_t id, MsgHandler_t handler) {
    if(handlerCount < MAX_HANDLERS) {
        handlers[handlerCount].id = id;
        handlers[handlerCount].handler = handler;
        handlerCount++;
    }
}
// ========== 通讯数据更新（协议模块调用） ==========
void MsgScheduler_UpdateCommData(const CommData_t *data) {
    commData = *data;   // 更新内部数据
    // 通知所有注册了 MSG_COMM_DATA_UPDATE 的模块
    for(uint8_t i = 0; i < handlerCount; i++) {
        if(handlers[i].id == MSG_COMM_DATA_UPDATE) {
            handlers[i].handler(NULL, &commData);   // 传数据指针
        }
    }
}
// ========== 消息发送 ==========
void Msg_Post(MsgId_t id, uint16_t param) {
    if(slotCount >= MSG_SLOT_DEPTH) return;  // 满了（实际不会发生）
    
    msgSlot[slotTail].id = id;
    msgSlot[slotTail].param = param;
    slotTail = (slotTail + 1) % MSG_SLOT_DEPTH;
    slotCount++;
}
// ========== 1ms调度入口 ==========
void MsgScheduler_Run1ms(void) {
    // 处理消息槽中的一条消息
    if(slotCount > 0) {
        Msg_t msg = msgSlot[slotHead];
        slotHead = (slotHead + 1) % MSG_SLOT_DEPTH;
        slotCount--;
        
        // 遍历注册表，匹配ID的回调全部调用
        for(uint8_t i = 0; i < handlerCount; i++) {
            if(handlers[i].id == msg.id) {
                handlers[i].handler(&msg, NULL);   // 普通消息不传额外数据
            }
        }
    }
}

// buzzer.c
#include "buzzer.h"
#include "hal_io.h"
static uint8_t  beepRemain;
static uint16_t beepInterval;
static uint16_t beepTimer;
void BUZZER_Init(void) {
    beepRemain = 0;
    beepInterval = 0;
    beepTimer = 0;
    HAL_Buzzer_Off();
}
// 唯一直接回调，任何模块直接调用
void BUZZER_Beep(uint8_t times, uint16_t intervalMs) {
    if(times == 0) return;
    beepRemain   = times - 1;
    beepInterval = intervalMs;
    beepTimer    = 0;
    HAL_Buzzer_On();
}
// 在1ms定时中断里调用
void BUZZER_Tick1ms(void) {
    if(beepRemain > 0 || HAL_Buzzer_IsOn()) {
        if(beepTimer >= beepInterval) {
            beepTimer = 0;
            if(beepRemain > 0) {
                HAL_Buzzer_On();
                beepRemain--;
            } else {
                HAL_Buzzer_Off();
            }
        }
        beepTimer++;
    }
}

// proto.c
#include "proto.h"
#include "msg_scheduler.h"
#include "hal_comm.h"
#define FRAME_HEADER 0xAA
// 接收状态机
static uint8_t rxBuf[16];
static uint8_t rxIndex;
static uint8_t rxState;   // 0=找帧头, 1=收数据
static uint8_t rxLen;
void PROTO_Init(void) {
    rxIndex = 0;
    rxState = 0;
    MsgScheduler_Register(MSG_POWER_CTRL, PROTO_OnSendPower);
}
// ========== 发送侧：MSG_POWER_CTRL 回调 ==========
void PROTO_OnSendPower(Msg_t *msg, void *dataPtr) {
    uint16_t power = msg->param;
    uint8_t buf[6];
    buf[0] = FRAME_HEADER;
    buf[1] = 0x01;              // 设备地址
    buf[2] = 0x10;              // 命令：设置功率
    buf[3] = (power >> 8) & 0xFF;
    buf[4] = power & 0xFF;
    buf[5] = buf[0] ^ buf[1] ^ buf[2] ^ buf[3] ^ buf[4];  // 简单校验
    HAL_Comm_Send(buf, 6);
}
// ========== 接收侧：中断里喂字节 ==========
void PROTO_OnByteReceived(uint8_t byte) {
    if(rxState == 0 && byte == FRAME_HEADER) {
        rxIndex = 0;
        rxBuf[rxIndex++] = byte;
        rxState = 1;
    } else if(rxState == 1) {
        rxBuf[rxIndex++] = byte;
        if(rxIndex >= 6) {
            uint8_t checksum = rxBuf[0] ^ rxBuf[1] ^ rxBuf[2] ^ rxBuf[3] ^ rxBuf[4];
            if(checksum == rxBuf[5]) {
                // 解包，更新到调度器内部数据区
                CommData_t data;
                data.igbtTemp    = (rxBuf[3] << 8) | rxBuf[4];
                data.surfaceTemp = 0;  // 根据实际协议填充
                data.coilCurrent = 0;
                data.systemState = 0;
                data.faultCode   = 0;
                data.commStatus  = 0;   // 正常
                MsgScheduler_UpdateCommData(&data);
            }
            rxState = 0;
        }
    }
}

// cook.c
#include "cook.h"
#include "msg_scheduler.h"
#include "buzzer.h"
// 菜谱表
typedef struct {
    uint16_t durationSec;
    uint16_t power;
    uint8_t  fanSpeed;
} CookStep_t;
static const CookStep_t menu1[] = {
    { 60, 2000, 3 },   // 预热60秒，2000W，风机3档
    { 120, 1500, 2 },  // 烹饪120秒，1500W，风机2档
    { 30,  800,  1 },  // 保温30秒，800W，风机1档
    { 0,   0,    0 },  // 结束标志
};
static uint8_t  cookingActive;
static uint8_t  currentStep;
static uint16_t stepTimer;  // 当前步骤已执行时间（秒）
void COOK_Init(void) {
    cookingActive = 0;
    currentStep = 0;
    stepTimer = 0;
    
    MsgScheduler_Register(MSG_COOKING_CTRL,     COOK_OnCookingCtrl);
    MsgScheduler_Register(MSG_TIMER_100MS,      COOK_OnTimer);
    MsgScheduler_Register(MSG_COMM_DATA_UPDATE, COOK_OnCommDataUpdate);
    MsgScheduler_Register(MSG_SYSTEM_ERROR,     COOK_OnSystemError);
}
void COOK_Task(void) {
    // 主循环中的烹饪状态机（如有需要）
}
// 烹饪控制回调
void COOK_OnCookingCtrl(Msg_t *msg, void *dataPtr) {
    uint8_t cmd = msg->param & 0xFF;
    
    if(cmd == 0x01) {  // 开始
        cookingActive = 1;
        currentStep = 0;
        stepTimer = 0;
        
        // 执行第一步：下发功率、开风机、开加热
        Msg_Post(MSG_POWER_CTRL, menu1[0].power);
        Msg_Post(MSG_FAN_CTRL, menu1[0].fanSpeed);
        Msg_Post(MSG_DISPLAY_REFRESH, menu1[0].durationSec);
        BUZZER_Beep(1, 100);
        
    } else if(cmd == 0x00) {  // 停止
        cookingActive = 0;
        Msg_Post(MSG_POWER_CTRL, 0);   // 关加热
        Msg_Post(MSG_FAN_CTRL, 0);     // 关风机
        Msg_Post(MSG_DISPLAY_REFRESH, 0);
        BUZZER_Beep(2, 150);
    }
}
// 100ms定时回调
void COOK_OnTimer(Msg_t *msg, void *dataPtr) {
    if(!cookingActive) return;
    
    // 每100ms调用一次，累加10次=1秒
    static uint8_t tickCount = 0;
    tickCount++;
    if(tickCount < 10) return;
    tickCount = 0;
    
    stepTimer++;
    
    const CookStep_t *step = &menu1[currentStep];
    if(step->durationSec == 0) {
        // 烹饪完成
        cookingActive = 0;
        Msg_Post(MSG_POWER_CTRL, 0);
        Msg_Post(MSG_FAN_CTRL, 0);
        Msg_Post(MSG_DISPLAY_REFRESH, 0x8888);  // 显示"完成"
        BUZZER_Beep(3, 200);
        return;
    }
    
    if(stepTimer >= step->durationSec) {
        // 进入下一步
        currentStep++;
        stepTimer = 0;
        step = &menu1[currentStep];
        
        if(step->durationSec == 0) return;  // 结束标志
        
        // 下发新步骤的参数
        Msg_Post(MSG_POWER_CTRL, step->power);
        Msg_Post(MSG_FAN_CTRL, step->fanSpeed);
        Msg_Post(MSG_DISPLAY_REFRESH, step->durationSec);
        BUZZER_Beep(1, 80);  // 阶段切换提示
    }
}
// 通讯数据更新回调
void COOK_OnCommDataUpdate(Msg_t *msg, void *dataPtr) {
    CommData_t *data = (CommData_t *)dataPtr;
    if(data->faultCode != 0) {
        // 电磁炉报故障，停止烹饪
        cookingActive = 0;
        Msg_Post(MSG_POWER_CTRL, 0);
        Msg_Post(MSG_SYSTEM_ERROR, data->faultCode);
    }
}
// 系统错误回调
void COOK_OnSystemError(Msg_t *msg, void *dataPtr) {
    cookingActive = 0;
    Msg_Post(MSG_POWER_CTRL, 0);
}

## 总结

### 架构核心

text

1ms定时中断：
  ├── BUZZER_Tick1ms()        // 蜂鸣器状态机（独立，不走消息）
  └── MsgScheduler_Run1ms()   // 取消息槽 → 匹配回调 → 执行
数据流：
  通讯中断 → PROTO_OnByteReceived()
           → 解包
           → MsgScheduler_UpdateCommData()  // 更新内部数据区
           → 回调 UI_OnCommDataUpdate / COOK_OnCommDataUpdate（传指针）
  按键中断 → Msg_Post(MSG_KEY_EVENT)
           → 1ms后调度器回调 UI_OnKeyEvent
           → UI 再发 MSG_COOKING_CTRL / MSG_FAN_CTRL
           → 调度器回调对应模块

### 关键原则

|通道|用途|实现|
|---|---|---|
|**消息槽**|所有业务事件|`Msg_Post` → 1ms后回调|
|**数据指针**|通讯共享数据|`MsgScheduler_UpdateCommData` → 直接传指针|
|**直接回调**|蜂鸣器|`BUZZER_Beep()`，任何模块直接调|
|**无全局变量**|数据隔离|所有共享数据在调度器内部|

这就是完整的项目框架，可以直接移植。各个硬件抽象层（`hal_*.c`）根据你的具体MCU实现即可。
